Авторизация function login() { const password = document.getElementById('pass').value; if (password === 'ara') { document.getElementById('loginScreen').classList.add('d-none'); document.getElementById('adminPanel').classList.remove('d-none'); loadData(); } else { alert('Неверный пароль!'); } } // 2. Форматирование даты (чтобы было красиво) function formatDate(dateString) { if (!dateString) return ""; // Если дата пришла в формате ISO (2026-02-28T...) if (dateString.includes("T")) { const d = new Date(dateString); return d.toLocaleDateString('ru-RU'); // Вернет 28.02.2026 } return dateString; } // 3. Загрузка данных из Google Sheets function loadData() { const tableBody = document.getElementById('adminTableBody'); tableBody.innerHTML = 'Загрузка данных...'; const url = "https://script.google.com/macros/s/AKfycbwOHbNcSKZ4R_G1iGJ510IuqL9gt6fBYM5EZeJBXcrc14UEuS2ZadU7eaS3zTT_zbKeTw/exec"; // Твоя ссылка fetch(url) .then(response => response.json()) .then(data => { tableBody.innerHTML = ''; // Пропускаем первую строку (заголовки) for (let i = 1; i < data.length; i++) { let row = data[i]; // Красиво оформляем школу let schoolName = row[4] ? row[4] : "Общая школа"; let schoolBadge = row[4] ? `${row[4]}` : `Не указана`; // Форматируем дату let dateFormatted = formatDate(row[0]); // Время (если оно пришло как 1899-12-30T...) let timeFormatted = row[1]; if (timeFormatted && timeFormatted.includes("T")) { let timeParts = timeFormatted.split("T")[1].split(":"); timeFormatted = timeParts[0] + ":" + timeParts[1]; } tableBody.innerHTML += ` ${schoolBadge} ${dateFormatted} ${timeFormatted} ${row[2]} ${row[3]} `; } if (data.length <= 1) { tableBody.innerHTML = 'Заявок пока нет.'; } }) .catch(() => { tableBody.innerHTML = 'Ошибка загрузки данных.'; }); }
